"use client";
import { useState, useEffect } from 'react';

export default function HackerTracker() {
  const [password, setPassword] = useState('');
  const [isAuthenticated, setIsAuthenticated] = useState(false);
  const [checking, setChecking] = useState(true);
  const [logs, setLogs] = useState<any[]>([]);
  const [loading, setLoading] = useState(true);
  const [filter, setFilter] = useState('all');

  useEffect(() => {
    const token = localStorage.getItem('admin_token');
    if (token) {
      verifyToken(token);
    } else {
      setChecking(false);
    }
  }, []);

  const verifyToken = async (token: string) => {
    try {
      const res = await fetch('/api/admin/verify-password', {
        method: 'POST',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify({ password: token }),
      });
      if (res.ok) {
        setIsAuthenticated(true);
        loadLogs();
      }
    } catch {} finally {
      setChecking(false);
    }
  };

  const handleLogin = async () => {
    const res = await fetch('/api/admin/verify-password', {
      method: 'POST',
      headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify({ password }),
    });
    if (res.ok) {
      localStorage.setItem('admin_token', password);
      setIsAuthenticated(true);
      loadLogs();
    } else {
      alert('Invalid admin password');
    }
  };

  const loadLogs = async () => {
    setLoading(true);
    try {
      const res = await fetch('/api/admin/hacker-traffic');
      if (res.ok) {
        const data = await res.json();
        setLogs(data.logs || []);
      }
    } catch (err) {
      console.error('Failed to load logs', err);
    } finally {
      setLoading(false);
    }
  };

  if (checking) {
    return (
      <div className="min-h-screen bg-[#020203] text-white flex items-center justify-center">
        <div className="animate-spin w-8 h-8 border-2 border-red-400 border-t-transparent rounded-full"></div>
      </div>
    );
  }

  if (!isAuthenticated) {
    return (
      <div className="min-h-screen bg-[#020203] text-white flex items-center justify-center">
        <div className="bg-[#07070a] border border-red-500/20 rounded-2xl p-8 max-w-md w-full mx-4">
          <h1 className="text-2xl font-bold mb-6 text-center text-transparent bg-clip-text bg-gradient-to-r from-red-400 to-yellow-500">
            HACKER TRACKER
          </h1>
          <p className="text-xs text-gray-500 text-center mb-4 font-mono">🔒 Admin authentication required</p>
          <input
            type="password"
            value={password}
            onChange={(e) => setPassword(e.target.value)}
            onKeyDown={(e) => e.key === 'Enter' && handleLogin()}
            placeholder="Admin Password..."
            className="w-full bg-white/5 border border-white/10 rounded-xl p-4 text-white placeholder:text-gray-600 focus:outline-none focus:border-red-500 mb-4"
          />
          <button
            onClick={handleLogin}
            className="w-full py-3 rounded-xl font-bold bg-gradient-to-r from-red-500 to-yellow-600 text-white hover:shadow-[0_0_30px_rgba(239,68,68,0.4)] transition-all"
          >
            ACCESS SECURE PANEL
          </button>
        </div>
      </div>
    );
  }

  const filteredLogs = filter === 'all' ? logs : logs.filter((l: any) => l.severity === filter);
  const severityColor = (s: string) => {
    switch (s) {
      case 'critical': return 'text-red-400 bg-red-500/10 border-red-500/30';
      case 'high': return 'text-orange-400 bg-orange-500/10 border-orange-500/30';
      case 'medium': return 'text-yellow-400 bg-yellow-500/10 border-yellow-500/30';
      case 'low': return 'text-cyan-400 bg-cyan-500/10 border-cyan-500/30';
      default: return 'text-gray-400 bg-gray-500/10 border-gray-500/30';
    }
  };

  return (
    <div className="min-h-screen bg-[#020203] text-white">
      {/* Top Nav */}
      <div className="bg-[#07070a] border-b border-white/5 px-6 py-4 flex items-center justify-between">
        <div className="flex items-center gap-4">
          <h1 className="text-lg font-black tracking-tight">
            <span className="text-transparent bg-clip-text bg-gradient-to-r from-red-400 to-yellow-500">HACKER</span> TRACKER
          </h1>
          <div className="h-4 w-[1px] bg-white/10"></div>
          <a href="/admin" className="text-xs text-gray-500 hover:text-white font-bold uppercase tracking-widest transition-colors">Dashboard</a>
          <a href="/admin/hacker-tracker" className="text-xs text-red-400 font-bold uppercase tracking-widest">Tracker</a>
          <a href="/admin/support" className="text-xs text-gray-500 hover:text-white font-bold uppercase tracking-widest transition-colors">Support</a>
        </div>
        <button
          onClick={() => { localStorage.removeItem('admin_token'); setIsAuthenticated(false); }}
          className="text-xs text-red-400 hover:text-red-300 font-bold uppercase tracking-widest"
        >
          Logout
        </button>
      </div>

      <div className="p-6 max-w-7xl mx-auto">
        {/* Stats Bar */}
        <div className="grid grid-cols-1 md:grid-cols-4 gap-4 mb-6">
          <div className="bg-[#07070a] border border-red-500/10 rounded-xl p-4">
            <p className="text-xs text-gray-500 uppercase tracking-widest font-mono">Total Events</p>
            <p className="text-2xl font-black text-white">{logs.length}</p>
          </div>
          <div className="bg-[#07070a] border border-red-500/10 rounded-xl p-4">
            <p className="text-xs text-gray-500 uppercase tracking-widest font-mono">Critical</p>
            <p className="text-2xl font-black text-red-400">{logs.filter((l: any) => l.severity === 'critical').length}</p>
          </div>
          <div className="bg-[#07070a] border border-red-500/10 rounded-xl p-4">
            <p className="text-xs text-gray-500 uppercase tracking-widest font-mono">High</p>
            <p className="text-2xl font-black text-orange-400">{logs.filter((l: any) => l.severity === 'high').length}</p>
          </div>
          <div className="bg-[#07070a] border border-red-500/10 rounded-xl p-4">
            <p className="text-xs text-gray-500 uppercase tracking-widest font-mono">Low/Info</p>
            <p className="text-2xl font-black text-cyan-400">{logs.filter((l: any) => ['low', 'info'].includes(l.severity)).length}</p>
          </div>
        </div>

        {/* Filter tabs */}
        <div className="flex gap-2 mb-6">
          {['all', 'critical', 'high', 'medium', 'low'].map((f) => (
            <button
              key={f}
              onClick={() => setFilter(f)}
              className={`px-4 py-1.5 rounded-lg text-xs font-bold uppercase tracking-widest transition-all ${
                filter === f
                  ? 'bg-red-500/20 text-red-400 border border-red-500/30'
                  : 'bg-white/5 text-gray-500 border border-white/10 hover:text-white'
              }`}
            >
              {f}
            </button>
          ))}
          <button
            onClick={loadLogs}
            className="px-4 py-1.5 rounded-lg text-xs font-bold uppercase tracking-widest bg-cyan-500/10 text-cyan-400 border border-cyan-500/20 hover:bg-cyan-500/20 ml-auto"
          >
            ↻ Refresh
          </button>
        </div>

        {/* Logs Table */}
        <div className="bg-[#07070a] border border-white/5 rounded-xl overflow-hidden">
          {loading ? (
            <div className="flex items-center justify-center py-20">
              <div className="animate-spin w-8 h-8 border-2 border-red-400 border-t-transparent rounded-full"></div>
            </div>
          ) : filteredLogs.length === 0 ? (
            <div className="text-center py-16">
              <div className="text-4xl mb-4">🛡️</div>
              <p className="text-gray-500 font-mono text-sm">No intrusion events detected</p>
              <p className="text-gray-600 text-xs mt-2">Your perimeter is secure, Overlord.</p>
            </div>
          ) : (
            <div className="overflow-x-auto">
              <table className="w-full text-sm">
                <thead>
                  <tr className="text-gray-500 text-xs uppercase tracking-widest border-b border-white/5">
                    <th className="text-left py-3 px-4">Time</th>
                    <th className="text-left py-3 px-4">Severity</th>
                    <th className="text-left py-3 px-4">IP</th>
                    <th className="text-left py-3 px-4">Event</th>
                    <th className="text-left py-3 px-4">Path</th>
                  </tr>
                </thead>
                <tbody>
                  {filteredLogs.map((log: any, i: number) => (
                    <tr key={i} className="border-b border-white/5 hover:bg-white/5 transition-colors">
                      <td className="py-3 px-4 font-mono text-xs text-gray-400">{log.timestamp || log.created_at}</td>
                      <td className="py-3 px-4">
                        <span className={`text-[10px] font-mono font-bold uppercase px-2 py-0.5 rounded border ${severityColor(log.severity)}`}>
                          {log.severity}
                        </span>
                      </td>
                      <td className="py-3 px-4 font-mono text-xs text-gray-300">{log.ip_address}</td>
                      <td className="py-3 px-4 text-gray-300">{log.event || log.event_type}</td>
                      <td className="py-3 px-4 font-mono text-xs text-gray-500">{log.path}</td>
                    </tr>
                  ))}
                </tbody>
              </table>
            </div>
          )}
        </div>
      </div>
    </div>
  );
}

import { NextResponse } from 'next/server';
import { supabase } from '@/lib/supabaseClient';

export const dynamic = 'force-dynamic';

export async function GET() {
  try {
    // Fetch stats from Supabase tables
    const [usersRes, creditsRes, paymentsRes, ticketsRes] = await Promise.all([
      supabase.from('user_credits').select('*'),
      supabase.from('user_credits').select('credits'),
      supabase.from('payments').select('*'),
      supabase.from('support_tickets').select('*').eq('status', 'open'),
    ]);

    const users = usersRes.data || [];
    const credits = creditsRes.data || [];
    const payments = paymentsRes.data || [];
    const openTickets = ticketsRes.data || [];

    const totalUsers = users.length;
    const activeCredits = credits.reduce((sum: number, u: any) => sum + (u.credits || 0), 0);
    const totalRevenue = payments
      .filter((p: any) => p.status === 'completed')
      .reduce((sum: number, p: any) => sum + (p.amount || 0), 0);

    return NextResponse.json({
      totalUsers,
      activeCredits,
      totalRevenue,
      openTickets: openTickets.length,
    });
  } catch (error) {
    console.error('Stats error:', error);
    return NextResponse.json({ totalUsers: 0, activeCredits: 0, totalRevenue: 0, openTickets: 0 });
  }
}

export async function POST(req: Request) {
  try {
    const { role, skills } = await req.json();
    if (!role || !skills) {
      return NextResponse.json({ error: "Job role and current skills are required" }, { status: 400 });
    }

    const apiKey = process.env.Gemini_Api_Key;
    if (!apiKey) {
      return NextResponse.json({ error: "API Key missing" }, { status: 500 });
    }

    const prompt = `You are an elite Tech Career Coach & Recruiter.
    Analyze the skill gap for a candidate targeting the role of "${role}".
    Their current skills/resume text: "${skills}".
    
    Provide a premium, highly detailed career roadmap in Markdown format. Include:
    1. Match Percentage (Give a prominent metric/percentage).
    2. Missing Core Skills (What they lack for this role).
    3. Missing Soft Skills/Tooling.
    4. Actionable 30-60-90 Day Learning Path to bridge this gap.
    Use professional markdown headers and clean bullet points.`;

    // ✅ SDK nahi, REST API — gemini-2.5-flash-lite free + working
    const geminiResponse = await fetch(
      `https://generativelanguage.googleapis.com/v1beta/models/gemini-2.5-flash-lite:generateContent?key=${apiKey}`,
      {
        method: 'POST',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify({
          contents: [{ parts: [{ text: prompt }] }]
        })
      }
    );

    if (!geminiResponse.ok) {
      const errData = await geminiResponse.json();
      throw new Error(errData.error?.message || "Gemini API failed");
    }

    const geminiData = await geminiResponse.json();
    const analysis = geminiData.candidates?.[0]?.content?.parts?.[0]?.text;

    if (!analysis) throw new Error("No response from Gemini");

    return NextResponse.json({ analysis });

  } catch (error: any) {
    console.error("Analyzer Error:", error);
    return NextResponse.json({ error: error.message || "Failed to analyze skills" }, { status: 500 });

import { Webhook } from 'svix';
import { headers } from 'next/headers';
import { NextResponse } from 'next/server';

export async function POST(req: Request) {
  const WEBHOOK_SECRET = process.env.CLERK_WEBHOOK_SECRET;

  if (!WEBHOOK_SECRET) {
    console.error('❌ CLERK_WEBHOOK_SECRET missing hai .env.local mein!');
    return new Response('Error', { status: 400 });
  }

  const headerPayload = await headers();
  const svix_id = headerPayload.get("svix-id");
  const svix_timestamp = headerPayload.get("svix-timestamp");
  const svix_signature = headerPayload.get("svix-signature");

  if (!svix_id || !svix_timestamp || !svix_signature) {
    console.error('❌ Clerk Headers missing hain!');
    return new Response('Error: Missing svix headers', { status: 400 });
  }

  // 🔥 FIX: req.json() ki jagah req.text() use karna zaroori hai signature ke liye
  const payload = await req.text();
  const wh = new Webhook(WEBHOOK_SECRET);
  let evt: any;

  try {
    evt = wh.verify(payload, {
      "svix-id": svix_id,
      "svix-timestamp": svix_timestamp,
      "svix-signature": svix_signature,
    });
  } catch (err) {
    console.error('❌ Signature Verification Failed:', err);
    return new Response('Error verifying webhook', { status: 400 });
  }

  // Agar yahan tak aaya, matlab Clerk ka message 100% sahi se receive ho gaya
  if (evt.type === 'user.created') {
    const { id } = evt.data;
    console.log(`🎉 NAYA USER AAYA HAI! ID: ${id}`);

    const supabaseUrl = process.env.NEXT_PUBLIC_SUPABASE_URL || '';
    const supabaseKey = process.env.SUPABASE_SERVICE_ROLE_KEY || '';

    try {
      const response = await fetch(`${supabaseUrl}/rest/v1/user_credits`, {
        method: 'POST',
        headers: {
          'apikey': supabaseKey,
          'Authorization': `Bearer ${supabaseKey}`,
          'Content-Type': 'application/json',
          'Prefer': 'return=minimal'
        },
        body: JSON.stringify({
          user_id: id,
          credits: 50,
          plan_name: 'Free Trial'
        })
      });

      if (!response.ok) {
        const errorText = await response.text();
        console.error("❌ SUPABASE REJECTED IT. Reason:", errorText);
      } else {
        console.log("✅ DATA SUPABASE MEIN CHALA GAYA!");
      }
    } catch (error) {
      console.error("❌ Fetch fail ho gaya:", error);
    }
  }

  return NextResponse.json({ message: 'Webhook Processed' }, { status: 200 });
}

"use client";
import { useEffect, useState } from "react";
import { motion } from "framer-motion";

export default function MouseTrail() {
  const [mousePosition, setMousePosition] = useState({ x: 0, y: 0 });

  useEffect(() => {
    const updateMousePosition = (e: MouseEvent) => {
      setMousePosition({ x: e.clientX, y: e.clientY });
    };
    window.addEventListener("mousemove", updateMousePosition);
    return () => {
      window.removeEventListener("mousemove", updateMousePosition);
    };
  }, []);

  return (
    <>
      <motion.div
        className="fixed top-0 left-0 w-8 h-8 border border-blue-500 rounded-full pointer-events-none z-[100] mix-blend-screen"
        animate={{ x: mousePosition.x - 16, y: mousePosition.y - 16 }}
        transition={{ type: "spring", stiffness: 150, damping: 15, mass: 0.5 }}
      />
      <motion.div
        className="fixed top-0 left-0 w-2 h-2 bg-emerald-400 rounded-full pointer-events-none z-[100] shadow-[0_0_10px_rgba(52,211,153,0.8)]"
        animate={{ x: mousePosition.x - 4, y: mousePosition.y - 4 }}
        transition={{ type: "spring", stiffness: 500, damping: 28, mass: 0.1 }}

    </>
  );
}

"use client";
import { useEffect, useRef, ReactNode } from 'react';

interface ScrollRevealProps {
  children: ReactNode;
  className?: string;
  delay?: number;
  direction?: 'up' | 'down' | 'left' | 'right' | 'none';
  duration?: number;
  threshold?: number;
}

export default function ScrollReveal({
  children,
  className = '',
  delay = 0,
  direction = 'up',
  duration = 700,
  threshold = 0.1,
}: ScrollRevealProps) {
  const ref = useRef<HTMLDivElement>(null);

  useEffect(() => {
    const el = ref.current;
    if (!el) return;

    const directionMap: Record<string, string> = {
      up: 'translateY(40px)',
      down: 'translateY(-40px)',
      left: 'translateX(-40px)',
      right: 'translateX(40px)',
      none: 'translateY(0)',
    };

    el.style.opacity = '0';
    el.style.transform = directionMap[direction];
    el.style.transition = `opacity ${duration}ms cubic-bezier(0.22, 1, 0.36, 1), transform ${duration}ms cubic-bezier(0.22, 1, 0.36, 1)`;
    el.style.transitionDelay = `${delay}ms`;

    const observer = new IntersectionObserver(
      ([entry]) => {
        if (entry.isIntersecting) {
          el.style.opacity = '1';
          el.style.transform = 'translateY(0) translateX(0)';
          observer.unobserve(el);
        }
      },
      { threshold, rootMargin: '0px 0px -50px 0px' }
    );

    observer.observe(el);
    return () => observer.disconnect();
  }, [delay, direction, duration, threshold]);

  return (
    <div ref={ref} className={className}>
      {children}
    </div>
  );
}

// PARTICLE SYSTEM
// ═══════════════════════════════════════════════════════════════════════
function ParticleField() {
  const canvasRef = useRef<HTMLCanvasElement>(null);
  const particlesRef = useRef<Particle[]>([]);
  const mouseRef = useRef({ x: -1000, y: -1000 });
  const rafRef = useRef<number>(0);

  useEffect(() => {
    const canvas = canvasRef.current;
    if (!canvas) return;
    const ctx = canvas.getContext("2d");
    if (!ctx) return;

    const resize = () => {
      const dpr = window.devicePixelRatio || 1;
      canvas.width = canvas.offsetWidth * dpr;
      canvas.height = canvas.offsetHeight * dpr;
      ctx.scale(dpr, dpr);
    };
    resize();
    window.addEventListener("resize", resize);

    const colors = ["#22d3ee", "#818cf8", "#34d399", "#a78bfa"];
    const particleCount = 50;

    particlesRef.current = Array.from({ length: particleCount }, () => ({
      x: Math.random() * canvas.offsetWidth,
      y: Math.random() * canvas.offsetHeight,
      vx: (Math.random() - 0.5) * 0.3,
      vy: (Math.random() - 0.5) * 0.3,
      size: Math.random() * 1.5 + 0.5,
      opacity: Math.random() * 0.4 + 0.1,
      color: colors[Math.floor(Math.random() * colors.length)],
    }));

    const handleMouseMove = (e: MouseEvent) => {
      const rect = canvas.getBoundingClientRect();
      mouseRef.current = { x: e.clientX - rect.left, y: e.clientY - rect.top };
    };
    const handleMouseLeave = () => {
      mouseRef.current = { x: -1000, y: -1000 };
    };

    canvas.addEventListener("mousemove", handleMouseMove);
    canvas.addEventListener("mouseleave", handleMouseLeave);

    const animate = () => {
      if (!ctx || !canvas) return;
      ctx.clearRect(0, 0, canvas.offsetWidth, canvas.offsetHeight);

      const particles = particlesRef.current;

      for (let i = 0; i < particles.length; i++) {
        const p = particles[i];

        const dx = p.x - mouseRef.current.x;
        const dy = p.y - mouseRef.current.y;
        const dist = Math.sqrt(dx * dx + dy * dy);
        if (dist < 100 && dist > 0) {
          const force = (100 - dist) / 100;
          p.vx += (dx / dist) * force * 0.5;
          p.vy += (dy / dist) * force * 0.5;
        }

        p.x += p.vx;
        p.y += p.vy;
        p.vx *= 0.98;
        p.vy *= 0.98;

        if (p.x < 0) p.x = canvas.offsetWidth;
        if (p.x > canvas.offsetWidth) p.x = 0;
        if (p.y < 0) p.y = canvas.offsetHeight;
        if (p.y > canvas.offsetHeight) p.y = 0;

        ctx.beginPath();
        ctx.arc(p.x, p.y, p.size, 0, Math.PI * 2);
        ctx.fillStyle = p.color;
        ctx.globalAlpha = p.opacity;
        ctx.fill();

        for (let j = i + 1; j < particles.length; j++) {
          const p2 = particles[j];
          const cdx = p.x - p2.x;
          const cdy = p.y - p2.y;
          const cdist = Math.sqrt(cdx * cdx + cdy * cdy);

          if (cdist < 80) {
            ctx.beginPath();
            ctx.moveTo(p.x, p.y);
            ctx.lineTo(p2.x, p2.y);
            ctx.strokeStyle = p.color;
            ctx.globalAlpha = (1 - cdist / 80) * 0.12;
            ctx.lineWidth = 0.5;
            ctx.stroke();
          }
        }
      }

      ctx.globalAlpha = 1;
      rafRef.current = requestAnimationFrame(animate);
    };

    animate();

    return () => {
      window.removeEventListener("resize", resize);
      canvas.removeEventListener("mousemove", handleMouseMove);
      canvas.removeEventListener("mouseleave", handleMouseLeave);
      cancelAnimationFrame(rafRef.current);
    };
  }, []);

  return (
    <canvas
      ref={canvasRef}
      className="absolute inset-0 w-full h-full pointer-events-auto"
      style={{ opacity: 0.5 }}
    />
  );
}

// TEXT SCRAMBLE
// ═══════════════════════════════════════════════════════════════════════
function ScrambleText({ text, className }: { text: string; className?: string }) {
  const [display, setDisplay] = useState(text);
  const chars = "ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789!@#$%^&*";

  const scramble = useCallback(() => {
    let iteration = 0;
    const interval = setInterval(() => {
      setDisplay(
        text
          .split("")
          .map((char, index) => {
            if (index < iteration) return text[index];
            if (char === " ") return " ";
            return chars[Math.floor(Math.random() * chars.length)];
          })
          .join("")
      );

      if (iteration >= text.length) clearInterval(interval);
      iteration += 0.5;
    }, 30);

    return () => clearInterval(interval);
  }, [text]);

  return (
    <span 
      className={className} 
      onMouseEnter={scramble}
      style={{ fontFamily: "ui-monospace, SFMono-Regular, Menlo, Monaco, Consolas, monospace" }}
    >
      {display}
    </span>
  );
}

// MAGNETIC BUTTON
// ═══════════════════════════════════════════════════════════════════════
function MagneticButton({ children, className }: { children: React.ReactNode; className?: string }) {
  const ref = useRef<HTMLDivElement>(null);
  const x = useMotionValue(0);
  const y = useMotionValue(0);

  const handleMouseMove = (e: React.MouseEvent) => {
    if (!ref.current) return;
    const rect = ref.current.getBoundingClientRect();
    const centerX = rect.left + rect.width / 2;
    const centerY = rect.top + rect.height / 2;
    x.set((e.clientX - centerX) * 0.15);
    y.set((e.clientY - centerY) * 0.15);
  };

  const handleMouseLeave = () => {
    x.set(0);
    y.set(0);
  };

  return (
    <motion.div
      ref={ref}
      className={className}
      style={{ x, y }}
      onMouseMove={handleMouseMove}
      onMouseLeave={handleMouseLeave}
      transition={{ type: "spring", stiffness: 150, damping: 15 }}
    >
      {children}
    </motion.div>
  );
}

// TILT CARD (Fixed - accepts style prop properly)
// ═══════════════════════════════════════════════════════════════════════
interface TiltCardProps {
  children: React.ReactNode;
  className?: string;
  cardStyle?: React.CSSProperties;
}

function TiltCard({ children, className, cardStyle }: TiltCardProps) {
  const ref = useRef<HTMLDivElement>(null);
  const [rotateX, setRotateX] = useState(0);
  const [rotateY, setRotateY] = useState(0);
  const [glowX, setGlowX] = useState(50);
  const [glowY, setGlowY] = useState(50);

  const handleMouseMove = (e: React.MouseEvent) => {
    if (!ref.current) return;
    const rect = ref.current.getBoundingClientRect();
    const x = (e.clientX - rect.left) / rect.width;
    const y = (e.clientY - rect.top) / rect.height;
    setRotateX((y - 0.5) * -10);
    setRotateY((x - 0.5) * 10);
    setGlowX(x * 100);
    setGlowY(y * 100);
  };

  const handleMouseLeave = () => {
    setRotateX(0);
    setRotateY(0);
    setGlowX(50);
    setGlowY(50);
  };

  const dynamicBg = `radial-gradient(circle at ${glowX}% ${glowY}%, rgba(34,211,238,0.06) 0%, transparent 60%)`;

  return (
    <motion.div
      ref={ref}
      className={className}
      onMouseMove={handleMouseMove}
      onMouseLeave={handleMouseLeave}
      style={{
        transform: `perspective(1000px) rotateX(${rotateX}deg) rotateY(${rotateY}deg)`,
        background: `${dynamicBg}, rgba(255,255,255,0.02)`,
        ...cardStyle,
      }}
      transition={{ type: "spring", stiffness: 300, damping: 30 }}
    >
      {children}
    </motion.div>
  );
}

-- EduAI Hacker Trap: Intrusion Detection System
-- Run this SQL in your Supabase SQL editor

-- 1. Create intrusion_logs table for tracking security events
CREATE TABLE IF NOT EXISTS intrusion_logs (
  id BIGSERIAL PRIMARY KEY,
  ip_address TEXT NOT NULL,
  path TEXT DEFAULT '/unknown',
  event_type TEXT NOT NULL,
  severity TEXT DEFAULT 'low' CHECK (severity IN ('low', 'medium', 'high', 'critical')),
  user_agent TEXT,
  headers JSONB,
  created_at TIMESTAMPTZ DEFAULT NOW()
);

-- Index for faster queries
CREATE INDEX IF NOT EXISTS idx_intrusion_logs_ip ON intrusion_logs(ip_address);
CREATE INDEX IF NOT EXISTS idx_intrusion_logs_severity ON intrusion_logs(severity);
CREATE INDEX IF NOT EXISTS idx_intrusion_logs_created ON intrusion_logs(created_at DESC);

-- 2. Create support_tickets table
CREATE TABLE IF NOT EXISTS support_tickets (
  id BIGSERIAL PRIMARY KEY,
  user_email TEXT,
  subject TEXT DEFAULT 'No Subject',
  message TEXT NOT NULL,
  status TEXT DEFAULT 'open' CHECK (status IN ('open', 'in_progress', 'resolved', 'closed')),
  created_at TIMESTAMPTZ DEFAULT NOW(),
  updated_at TIMESTAMPTZ DEFAULT NOW()
);

CREATE INDEX IF NOT EXISTS idx_support_tickets_status ON support_tickets(status);

-- 3. Create feedbacks table
CREATE TABLE IF NOT EXISTS feedbacks (
  id BIGSERIAL PRIMARY KEY,
  user_id TEXT DEFAULT 'anonymous',
  rating INTEGER DEFAULT 0,
  message TEXT,
  created_at TIMESTAMPTZ DEFAULT NOW()
);

-- 4. Create payments table (for Razorpay transactions)
CREATE TABLE IF NOT EXISTS payments (
  id BIGSERIAL PRIMARY KEY,
  user_id TEXT,
  order_id TEXT UNIQUE,
  payment_id TEXT,
  amount INTEGER,
  currency TEXT DEFAULT 'INR',
  status TEXT DEFAULT 'pending',
  plan_name TEXT,
  credits_purchased INTEGER DEFAULT 0,
  created_at TIMESTAMPTZ DEFAULT NOW(),
  updated_at TIMESTAMPTZ DEFAULT NOW()
);

CREATE INDEX IF NOT EXISTS idx_payments_user ON payments(user_id);
CREATE INDEX IF NOT EXISTS idx_payments_status ON payments(status);

-- 5. Create user_history table
CREATE TABLE IF NOT EXISTS user_history (
  id BIGSERIAL PRIMARY KEY,
  user_id TEXT,
  tool_name TEXT,
  input JSONB,
  output TEXT,
  credits_used INTEGER DEFAULT 1,
  created_at TIMESTAMPTZ DEFAULT NOW()
);

CREATE INDEX IF NOT EXISTS idx_user_history_user ON user_history(user_id);
CREATE INDEX IF NOT EXISTS idx_user_history_created ON user_history(created_at DESC);

-- 6. Create anonymous_usage table for free tier tracking
CREATE TABLE IF NOT EXISTS anonymous_usage (
  id BIGSERIAL PRIMARY KEY,
  ip_address TEXT,
  tool_name TEXT,
  date DATE DEFAULT CURRENT_DATE,
  count INTEGER DEFAULT 1,
  UNIQUE(ip_address, tool_name, date)
);

-- 7. Enable Row Level Security
ALTER TABLE intrusion_logs ENABLE ROW LEVEL SECURITY;
ALTER TABLE support_tickets ENABLE ROW LEVEL SECURITY;
ALTER TABLE feedbacks ENABLE ROW LEVEL SECURITY;
ALTER TABLE payments ENABLE ROW LEVEL SECURITY;
ALTER TABLE user_history ENABLE ROW LEVEL SECURITY;
ALTER TABLE anonymous_usage ENABLE ROW LEVEL SECURITY;

-- 8. Create policies (allow all authenticated access for now)
CREATE POLICY "Allow all on intrusion_logs" ON intrusion_logs FOR ALL USING (true) WITH CHECK (true);
CREATE POLICY "Allow all on support_tickets" ON support_tickets FOR ALL USING (true) WITH CHECK (true);
CREATE POLICY "Allow all on feedbacks" ON feedbacks FOR ALL USING (true) WITH CHECK (true);
CREATE POLICY "Allow all on payments" ON payments FOR ALL USING (true) WITH CHECK (true);
CREATE POLICY "Allow all on user_history" ON user_history FOR ALL USING (true) WITH CHECK (true);
CREATE POLICY "Allow all on anonymous_usage" ON anonymous_usage FOR ALL USING (true) WITH CHECK (true);
